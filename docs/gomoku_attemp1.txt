static void play_game(uint8_t level) {
 uint8_t turn;
 uint8_t input;
 setup_game();
 if (g_n == 0) {
  show_game_over();
  return;
 }
 draw_initial_board();
 turn = HUMAN;
 g_winner = 0;
 g_draw = 0;
 while (g_winner == 0 && g_draw == 0) {
  if (turn == HUMAN) {
   input = _XL_INPUT();
   if (_XL_LEFT(input)) {
    move_cursor(0, -1);
   } else if (_XL_RIGHT(input)) {
    move_cursor(0, 1);
   } else if (_XL_UP(input)) {
    move_cursor(-1, 0);
   } else if (_XL_DOWN(input)) {
    move_cursor(1, 0);
   } else if (_XL_FIRE(input)) {
    if (g_board[g_cur_r][g_cur_c] == EMPTY) {
     place_human();
     turn = AI;
    }
   }
  } else {
   ai_move(level);
   if (g_ai_r < g_n && g_ai_c < g_n && g_board[g_ai_r][g_ai_c] == EMPTY) {
    place_ai();
    turn = HUMAN;
   } else {
    g_draw = 1;
   }
  }
  _XL_SLOW_DOWN(_XL_SLOW_DOWN_FACTOR);
 }
 show_game_over();
}
